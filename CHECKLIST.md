## Quick Start: Add API to React Component
```tsx
import { useGetApiItems } from '@/generated/api/items/items';

export function ItemsPage() {
  const { data: items, isLoading } = useGetApiItems({
    householdId: 'household-id'
  });
  
  return (
    <div>
      {isLoading ? 'Loading...' : (
        <ul>
          {items?.map(item => (
            <li key={item.id}>{item.name}</li>
          ))}
        </ul>
      )}
    </div>
  );
}
```
